````typescript
import {Node, resources, Sprite, SpriteFrame, Texture2D, Vec3} from 'cc';
import {getPageTheme, PAGE_THEMES, PageStyle, PageTheme} from '../data/PageTheme';
import {
  drawRoundRect, makeButton, makeBodyLabel, makePanelHeading, transform,
} from './HotPotUi';
import {HotPotTools} from './HotPotTools';

/** Four approved skins live inside My; no game state is recreated. */
export abstract class HotPotAppearanceMenus extends HotPotTools {
  private styleOriginal: PageStyle | null = null;

  protected cancelPageStyle(): void {
    this.discardPageStylePreview();
    this.dismissPlayerMenu('StyleOverlay');
    this.showPlayerProfile();
  }

  protected discardPageStylePreview(): void {
    if (this.styleOriginal !== null) this.previewPageStyle(this.styleOriginal);
    this.styleOriginal = null;
  }

  protected showPageStyles(): void {
    const panel = this.makePlayerMenu('StyleOverlay', 1040);
    if (!panel) {
      return;
    }
    makePanelHeading(panel, '页面风格', 420, 420);
    if (this.styleOriginal === null) this.styleOriginal = getPageTheme().id;
    makeBodyLabel(panel, '点选预览 · 确认后应用 · 保留当前这锅',
      21, '#94735D', new Vec3(0, 353), getPageTheme().id === 8 ? 526 : 540, 38);
    const current = getPageTheme();
    PAGE_THEMES.forEach((theme, index) => {
      const selected = theme.id === current.id;
      const card = makeButton(panel, '',
        new Vec3(index % 2 ? 138 : -138, index < 2 ? 180 : -106),
        250, 264, current.panel, () => {
          this.previewPageStyle(theme.id);
          this.dismissPlayerMenu('StyleOverlay');
          this.showPageStyles();
        });
      card.name = `PageStyle_${theme.id}`;
      drawRoundRect(card, 250, 264, 18,
        current.card, selected ? current.edge : current.line, selected ? 4 : 1.5);
      const picture = new Node('StyleThumbnail');
      card.addChild(picture);
      picture.setPosition(-66, 16);
      drawRoundRect(picture, 84, 178, 7, theme.accent);
      this.loadStyleThumbnail(picture, theme);
      makeBodyLabel(card, theme.name, 23, current.ink,
        new Vec3(55, 53), 100, 40, true).isBold = true;
      makeBodyLabel(card, theme.caption.replace(' · ', '\n'),
        18, '#806C57', new Vec3(55, -6), 100, 56, true);
      makeBodyLabel(card, selected ? '✓ 已选中' : '点击预览',
        21, current.ink, new Vec3(0, -96), 210, 28);
    });
    makeBodyLabel(panel, `预览：${current.name}`,
      25, current.ink, new Vec3(0, -276), 510, 42);
    makeBodyLabel(panel, '食材共享，牌框与界面随风格切换',
      21, '#94735D', new Vec3(0, -318), current.id === 8 ? 510 : 530, 32, true);
    makeButton(panel, '应用所选风格', new Vec3(0, -382),
      510, 70, '#A3613B', () => {
        this.selectPageStyle(getPageTheme().id);
        this.styleOriginal = null;
        this.dismissPlayerMenu('StyleOverlay');
        this.showPlayerProfile();
      });
    makeButton(panel, '返回我的', new Vec3(0, -464),
      510, 66, '#B88261', () => {
        this.cancelPageStyle();
      });
  }

  private loadStyleThumbnail(node: Node, theme: PageTheme): void {
    if (!resources?.load) {
      return;
    }
    resources.load(theme.thumbnail + '/texture', Texture2D,
      (error, texture) => {
        if (error || !texture || !node.isValid) {
          return;
        }
        const frame = new SpriteFrame();
        frame.texture = texture;
        const image = new Node('Artwork');
        node.addChild(image);
        const sprite = image.addComponent(Sprite);
        sprite.sizeMode = Sprite.SizeMode.CUSTOM;
        transform(image, 84, 178);
        sprite.spriteFrame = frame;
        node.on(Node.EventType.NODE_DESTROYED, () => frame.destroy());
      });
  }

  protected abstract makePlayerMenu(
    name: string, height: number,
  ): Node | null;
  protected abstract dismissPlayerMenu(name: string): void;
}
````

## assets/scripts/core/HotPotBoard.ts

````typescript
import {Graphics, Node, tween, Vec3} from 'cc';

import {getIngredient, IngredientType} from '../data/GameData';

import {
  drawFoodTile,
  drawTavernPanel,
  hex,
  makeLabel,
  opacity,
  transform,
  TRAY_IMAGE_HEIGHT,
  TRAY_IMAGE_WIDTH,
  TRAY_SPACING,
  TRAY_START_X,
} from './HotPotUi';

import {HotPotBoardActions} from './HotPotBoardActions';

/**
 * HotPotBoard：棋盘、托盘与消除。
 */
export abstract class HotPotBoard extends HotPotBoardActions {
  /**
   * 播放三消和后续托盘收拢动画。
   */
  protected animateMatch(matchStart: number): void {
    this.playEffect('match', 0.82);
    const trayItems = this.trayNode.children.filter(
      /**
       * 按条件筛选或判断条目：child.name.startsWith('TrayItem_')。
       */ (child) => child.name.startsWith('TrayItem_'),
    );
    const matching = trayItems.slice(matchStart, matchStart + 3);
    const effects = new Node('MatchEffects');
    this.trayNode.addChild(effects);
    // Commit the match before allowing another click. Animation nodes no
    // longer occupy slots and cannot modify a later tray or restarted game.
    this.tray.splice(matchStart, 3);
    matching.forEach(
      /**
       * 依次处理集合中的每个条目。
       */ (item) => {
        item.setParent(effects);
        const start = item.position.clone();
        tween(item)
          .to(0.08, {scale: new Vec3(1.12, 0.78, 1)})
          .to(
            0.18,
            {
              scale: new Vec3(1.2, 1.2, 1),
            },
            {easing: 'backOut'},
          )
          .delay(0.08)
          .to(
            0.22,
            {
              scale: new Vec3(0.02, 0.02, 1),
            },
            {easing: 'cubicIn'},
          )
          .start();
        tween(opacity(item)).delay(0.34).to(0.22, {opacity: 0}).start();
        this.createTrayMatchBurst(start.x, effects);
      },
    );

    // 与消除同时开始：直接移动右侧的真实食材节点，不创建副本、不保留原图。
    trayItems.slice(matchStart + 3).forEach(
      /**
       * 依次处理集合中的每个条目。
       */ (item, followerOffset) => {
        const trayIndex = matchStart + 3 + followerOffset;
        const targetX = TRAY_START_X + (trayIndex - 3) * TRAY_SPACING;
        tween(item)
          .to(
            0.78,
            {
              position: new Vec3(targetX, 0),
              scale: Vec3.ONE,
            },
            {easing: 'sineInOut'},
          )
          .start();
      },
    );
    const reward = this.grantBoilRewards();
    this.updateProgress();
    if (reward) this.showToast(reward, '#D64A36');
    tween(effects)
      .delay(0.82)
      .call(
        /**
         * 在动画到达指定阶段后更新游戏状态。
         */ () => effects.destroy(),
      )
      .start();
    if (this.isBoardCleared()) {
      this.endGame(true);
    } else {
      this.locked = false;
      this.checkpointGame();
    }
  }

  /**
   * 移除上一轮的匹配特效。
   */
  protected clearMatchEffects(): void {
    this.trayNode.children
      .slice()
      .filter(
        /**
         * 按条件筛选或判断条目：child.name === 'MatchEffects'。
         */ (child) => child.name === 'MatchEffects',
      )
      .forEach(
        /**
         * 依次处理集合中的每个条目。
         */ (child) => {
          child.removeFromParent();
          child.destroy();
        },
      );
  }

  /**
   * 星芒从每张牌的原位向四周散开，三个节点沿同一时间轴同步消除。
   */
  protected createTrayMatchBurst(centerX: number, parent: Node): void {
    const colors = ['#FFD66B', '#FF8A52', '#FFF1B8'];
    for (let i = 0; i < 16; i += 1) {
      const particle = new Node(`MatchParticle_${i}`);
      parent.addChild(particle);
      particle.setPosition(centerX, 0);
      transform(particle, 22, 22);
      const graphics = particle.addComponent(Graphics);
      graphics.fillColor = hex(colors[i % colors.length]);
      const radius = i % 3 === 0 ? 13 : 8;
      for (let point = 0; point < 8; point += 1) {
        const angle = (Math.PI * point) / 4;
        const length = point % 2 === 0 ? radius : radius * 0.3;
        const x = Math.cos(angle) * length;
        const y = Math.sin(angle) * length;
        if (point === 0) {
          graphics.moveTo(x, y);
        } else {
          graphics.lineTo(x, y);
        }
      }
      graphics.close();
      graphics.fill();

      const angle = (Math.PI * 2 * i) / 16;
      const distance = 80 + (i % 4) * 18;
      particle.setScale(0, 0, 1);
      tween(particle)
        .delay(0.26)
        .set({scale: Vec3.ONE})
        .to(
          0.5,
          {
            position: new Vec3(
              Math.max(
                -340,
                Math.min(340, centerX + Math.cos(angle) * distance),
              ),
              Math.sin(angle) * distance,
            ),
            scale: new Vec3(0.12, 0.12, 1),
            angle: i % 2 === 0 ? 150 : -150,
          },
          {easing: 'quadOut'},
        )
        .call(
          /**
           * 在动画到达指定阶段后更新游戏状态。
           */ () => particle.destroy(),
        )
        .start();
      tween(opacity(particle)).delay(0.4).to(0.36, {opacity: 0}).start();
    }
  }

  /**
   * 依据托盘数据重新绘制食材节点。
   */
  protected renderTray(): void {
    this.updateProgress();
    // destroy 在帧末才生效，先移出托盘，确保消除动画只选中本次创建的三个节点。
    this.trayNode.children.slice().forEach(
      /**
       * 依次处理集合中的每个条目。
       */ (child) => {
        if (child.name === 'MatchEffects') {
          return;
        }
        child.removeFromParent();
        child.destroy();
      },
    );

    // 第一层只绘制固定槽位，补位时边框保持不动。
    for (let i = 0; i < this.levelData.config.traySize; i += 1) {
      const slot = new Node(`TraySlot_${i}`);
      this.trayNode.addChild(slot);
      slot.setPosition(TRAY_START_X + i * TRAY_SPACING, 0);
      drawTavernPanel(slot, 86, 94, 'slot');
    }

    // 第二层放真实食材节点；全部位于固定槽位之上，并可直接执行左移动画。
    for (let i = 0; i < this.tray.length; i += 1) {
      const type = this.tray[i];
      const item = new Node(`TrayItem_${i}`);
      this.trayNode.addChild(item);
      item.setPosition(TRAY_START_X + i * TRAY_SPACING, 0);
      transform(item, 82 * 1.2, 84 * 1.2);
      drawFoodTile(item, 80, 78);
      (
        item as Node & {
          /**
           * trayType 对应的状态或配置。
           */ trayType?: IngredientType;
        }
      ).trayType = type;
      const ingredient = getIngredient(type);
      if (this.ingredientFrames.has(type)) {
        this.addIngredientImage(
          item,
          type,
          false,
          TRAY_IMAGE_WIDTH,
          TRAY_IMAGE_HEIGHT,
        );
      } else {
        makeLabel(item, '?', 32, ingredient.dark, Vec3.ZERO, 56, 48);
      }
    }
  }

  /**
   * 统计棋盘上尚未移除的牌。
   */
  protected remainingTiles(): number {
    return this.levelData.tiles.filter(
      /**
       * 按条件筛选或判断条目：!tile.removed。
       */ (tile) => !tile.removed,
    ).length;
  }

  /**
   * 判断棋盘是否已经清空。
   */
  protected isBoardCleared(): boolean {
    return (
      this.remainingTiles() === 0 &&
      this.tray.length === 0 &&
      this.reserve.length === 0
    );
  }

  /**
   * 执行复活并恢复可继续操作的状态。
   */
  protected revive(): void {
    if (!this.gameEnded
      || this.reviveUsed ||
      this.tray.length < this.levelData.config.traySize) {
      return;
    }
    this.reviveUsed = true;
    this.clearMatchEffects();
    this.reserve = this.tray.splice(0, 3);
    this.lastMoveSnapshot = null;
    this.dismissOverlay();
    this.gameEnded = false;
    this.locked = false;
    this.combo = 0;
    this.renderTray();
    this.renderReserve();
    this.checkpointGame();
    this.showToast('已复活！寄存食材可点击放回托盘', '#D86A3B');
  }

  /**
   * 将暂存区的食材取回托盘。
   */
  protected takeReservedTile(index: number): void {
    if (this.locked
      || this.gameEnded ||
      this.hasPlayerMenu() ||
      index < 0 ||
      index >= this.reserve.length) {
      return;
    }
    this.lastMoveSnapshot = {
      tray: this.tray.slice(),
      reserve: this.reserve.slice(),
      removedIds: this.levelData.tiles
        .filter(
          /**
           * 按条件筛选或判断条目：tile.removed。
           */ (tile) => tile.removed,
        )
        .map(
          /**
           * 将当前条目转换为后续处理所需的值。
           */ (tile) => tile.id,
        ),
    };
    this.locked = true;
    const type = this.reserve.splice(index, 1)[0];
    this.insertIntoTray(type);
    this.renderReserve();
    this.resolveTray();
  }

  /**
   * 重新绘制暂存区的食材。
   */
  protected renderReserve(): void {
    this.reserveNode.children.slice().forEach(
      /**
       * 依次处理集合中的每个条目。
       */ (child) => {
        child.removeFromParent();
        child.destroy();
      },
    );
    this.reserveNode.active = this.reserve.length > 0;
    // Make room above the tray instead of covering playable board tiles.
    const scale = this.reserve.length > 0 ? 0.86 : 1;
    this.board.setScale(scale, scale, 1);
    if (!this.reserve.length) {
      return;
    }
    drawTavernPanel(this.reserveNode, 560, 90, 'wood');
    makeLabel(
      this.reserveNode,
      '寄存食材\n点击取回',
      22,
      '#FFF8E7',
      new Vec3(-174, 0),
      170,
      66,
      true,
    );
    this.reserve.forEach(
      /**
       * 依次处理集合中的每个条目。
       */ (type, index) => {
        const item = new Node(`Reserved_${index}`);
        this.reserveNode.addChild(item);
        item.setPosition(-20 + index * 96, 0);
        item.setScale(0.75, 0.75, 1);
        this.drawTile(item, type, false);
        item.on(
          Node.EventType.TOUCH_END,
          /**
           * 响应节点触摸或游戏事件。
           */ () => this.takeReservedTile(index),
        );
      },
    );
  }
}
````

## assets/scripts/core/HotPotBoardActions.ts

````typescript
import {Color, Node, Sprite, tween, UITransform, Vec3} from 'cc';

import {
  createLevel,
  getIngredient,
  IngredientType,
  LevelData,
  LEVELS,
  TileData,
} from '../data/GameData';

import {
  drawFoodTile,
  EIGHT_ROW_TILE_SCALE,
  hex,
  makeLabel,
  opacity,
  TILE_HEIGHT,
  TILE_IMAGE_HEIGHT,
  TILE_IMAGE_WIDTH,
  TILE_WIDTH,
  transform,
  TRAY_SPACING,
  TRAY_START_X,
} from './HotPotUi';

import {HotPotView} from './HotPotView';
import {CLIENT_BUILD} from './HotPotTileSkin';

/**
 * HotPotBoardActions：拆分后的前段场景行为。
 */
export abstract class HotPotBoardActions extends HotPotView {
  /**
   * 生成并显示指定关卡，按需保存。
   */
  protected loadLevel(
    index: number,
    persist = true,
    restored?: LevelData,
  ): void {
    const loadStartedAt = Date.now();
    this.runId = `${Date.now().toString(36)}-`
      + Math.random().toString(36).slice(2, 14);
    this.clearMatchEffects();
    const previousLevel = this.levelData;
    this.levelIndex = Math.max(0, Math.min(LEVELS.length - 1, index));
    this.levelData = restored || createLevel(
      this.levelIndex,
      undefined,
      previousLevel?.config.level === 3 ? previousLevel.layoutSeed : undefined,
    );
    this.tray = [];
    this.reserve = [];
    this.reviveUsed = false;
    this.rewardedMilestones = 0;
    this.renderReserve();
    this.tileNodes.clear();
    this.locked = false;
    this.gameEnded = false;
    this.lastResult = 'playing';
    this.combo = 0;
    this.lastMoveSnapshot = null;
    const toolsEnabled = this.levelIndex >= 1;
    this.unlimitedTools = {undo: false, bomb: false, shuffle: false};
    this.toolUses = toolsEnabled
      ? {undo: 1, bomb: 1, shuffle: 1}
      : {undo: 0, bomb: 0, shuffle: 0};
    this.toolBar.active = toolsEnabled;
    this.updateToolButtons();
    this.effectSource?.stop();
    this.dismissOverlay();
    this.dismissRestartConfirmation();
    this.board.children.slice().forEach(
      /**
       * 依次处理集合中的每个条目。
       */ (child) => {
        if (child.name.startsWith('Tile_')) {
          child.removeFromParent();
          child.destroy();
        }
      },
    );
    this.foodTransitionLayer.children.slice().forEach(
      /**
       * 依次处理集合中的每个条目。
       */ (child) => child.destroy(),
    );
    this.levelLabel.string =
      '第 ' +
      String(this.levelData.config.level) +
      ' / ' +
      String(LEVELS.length) +
      ' 关  ·  ' +
      String(this.levelData.config.title) +
      '';
    this.refreshTileStates();
    this.renderTray();
    this.updateProgress();
    if (persist) {
      this.checkpointGame();
    }
    console.info(`[HotPot ${CLIENT_BUILD}] level=${this.levelIndex + 1}`
      + ` nodes=${this.tileNodes.size} loadMs=${Date.now() - loadStartedAt}`);
    // this.showToast('看清叠层，别让托盘塞满！', '#8A4B34');
  }

  /**
   * 为一张食材牌创建可点击的显示节点。
   */
  protected createTileNode(tile: TileData): void {
    const node = new Node(`Tile_${tile.id}`);
    this.board.addChild(node);
    node.setPosition(tile.x, tile.y, tile.layer);
    const tileScale =
      this.levelData.config.tileScale ||
      (this.levelData.config.rows === 8 ? EIGHT_ROW_TILE_SCALE : 1);
    node.setScale(tileScale, tileScale, 1);
    transform(node, TILE_WIDTH, TILE_HEIGHT);
    node.on(
      Node.EventType.TOUCH_END,
      /**
       * 响应节点触摸或游戏事件。
       */ () => this.onTileClicked(tile),
    );
    this.tileNodes.set(tile.id, node);
  }

  /**
   * 根据食材类型和遮挡状态绘制牌面。
   */
  protected drawTile(node: Node, type: IngredientType, blocked: boolean): void {
    node.children.slice().forEach(
      /**
       依次处理集合中的每个条目。
       */
      (child) => {
        child.removeFromParent();
        child.destroy();
      },
    );
    const ingredient = getIngredient(type);
    drawFoodTile(node, TILE_WIDTH, TILE_HEIGHT, blocked);
    if (this.ingredientFrames.has(type)) {
      this.addIngredientImage(
        node,
        type,
        blocked,
        TILE_IMAGE_WIDTH,
        TILE_IMAGE_HEIGHT,
      );
    } else {
      makeLabel(
        node,
        ingredient.name,
        26,
        blocked ? '#6F6255' : ingredient.dark,
        Vec3.ZERO,
        96,
        54,
      );
    }
    opacity(node, 255);
  }

  /**
   * 把食材图片裁切并放进牌面。
   */
  protected addIngredientImage(
    parent: Node,
    type: IngredientType,
    blocked: boolean,
    width: number,
    height: number,
  ): void {
    const frame = this.ingredientFrames.get(type);
    if (!frame) {
      return;
    }
    const image = new Node('IngredientImage');
    parent.addChild(image);
    // 按裁切后的资源比例等比适配，避免方形手绘图标被压扁。
    const scale = Math.min(
      width / frame.rect.width,
      height / frame.rect.height,
    );
    transform(image, frame.rect.width * scale, frame.rect.height * scale);
    const sprite = image.addComponent(Sprite);
    sprite.sizeMode = Sprite.SizeMode.CUSTOM;
    sprite.spriteFrame = frame;
    // 中性明暗保留食材本身的色相，避免全部染成黄褐色后难以辨认。
    sprite.color = blocked ? hex('#737373') : Color.WHITE;
  }

  /**
   * 按编号查找棋盘食材牌。
   */
  protected findTile(id: string): TileData | undefined {
    return this.levelData.tiles.find(
      /**
       * 按条件筛选或判断条目：tile.id === id。
       */ (tile) => tile.id === id,
    );
  }

  /**
   * 判断牌是否未移除且未被上层牌遮挡。
   */
  protected isTileSelectable(
    tile: TileData,
    tilesById?: ReadonlyMap<string, TileData>,
  ): boolean {
    if (tile.removed) {
      return false;
    }
    return !tile.blockedBy.some(
      /**
       * 按条件筛选或判断条目：{ const blocker = tilesById ? tilesById.get(id) :
       * this.findT。
       */ (id) => {
        const blocker = tilesById ? tilesById.get(id) : this.findTile(id);
        return blocker && !blocker.removed;
      },
    );
  }

  /**
   * 刷新全部牌的可选状态和外观。
   */
  protected refreshTileStates(): void {
    const tilesById = new Map(
      this.levelData.tiles.map(
        /**
         * 将当前条目转换为后续处理所需的值。
         */ (tile) => [tile.id, tile] as const,
      ),
    );
    // Repeated layers share exact positions. Only the highest remaining tile
    // at each position needs a node and sprites. Keep all
    // logical tiles, including those affected by bombs and shuffling.
    const frontTiles = new Map<string, TileData>();
    for (const tile of this.levelData.tiles) {
      if (tile.removed) {
        continue;
      }
      const key = `${tile.x}:${tile.y}`;
      const front = frontTiles.get(key);
      if (!front || tile.layer > front.layer) {
        frontTiles.set(key, tile);
      }
    }
    let sibling = 0;
    this.levelData.tiles.forEach(
      /**
       * 依次处理集合中的每个条目。
       */ (tile) => {
        if (tile.removed) {
          return;
        }
        const visible = frontTiles.get(`${tile.x}:${tile.y}`) === tile;
        if (visible && !this.tileNodes.has(tile.id)) {
          this.createTileNode(tile);
        }
        const node = this.tileNodes.get(tile.id);
        if (!node) {
          return;
        }
        node.active = visible;
        if (!node.active) {
          return;
        }
        // A newly exposed lower tile must remain behind higher layers.
        node.setSiblingIndex(sibling++);
        const selectable = this.isTileSelectable(tile, tilesById);
        if ((
            node as Node & {
              /**
               * selectableState 对应的状态或配置。
               */ selectableState?: boolean;
            }
          ).selectableState !== selectable) {
          (
            node as Node & {
              /**
               * selectableState 对应的状态或配置。
               */ selectableState?: boolean;
            }
          ).selectableState = selectable;
          this.drawTile(node, tile.type, !selectable);
        }
      },
    );
  }

  /**

      return;
    }
    const node = this.tileNodes.get(tile.id);
    if (!node) {
      return;
    }
    if (!this.isTileSelectable(tile)) {
      const start = node.position.clone();
      tween(node)
        .to(0.04, {position: new Vec3(start.x + 5, start.y, start.z)})
        .to(0.04, {position: new Vec3(start.x - 5, start.y, start.z)})
        .to(0.04, {position: start})
        .start();
      this.showToast('这张牌还被上层压着', '#A24A3C');
      return;
    }

    this.lastMoveSnapshot = {
      tray: this.tray.slice(),
      reserve: this.reserve.slice(),
      removedIds: this.levelData.tiles
        .filter(
          /**
           * 按条件筛选或判断条目：item.removed。
           */ (item) => item.removed,
        )
        .map(
          /**
           * 将当前条目转换为后续处理所需的值。
           */ (item) => item.id,
        ),
    };
    this.locked = true;
    tile.removed = true;
    const targetIndex = this.insertIntoTray(tile.type);
    const targetX = TRAY_START_X + Math.min(targetIndex, 6) * TRAY_SPACING;
    const trayUI = this.trayNode.getComponent(UITransform)!;
    const boardUI = this.board.getComponent(UITransform)!;
    const transitionUI = this.foodTransitionLayer.getComponent(UITransform)!;
    const startWorld = boardUI.convertToWorldSpaceAR(node.position);
    const targetWorld = trayUI.convertToWorldSpaceAR(new Vec3(targetX, 0));
    const startWorldScale = node.worldScale.clone();
    const transitionWorldScale = this.foodTransitionLayer.worldScale;
    const trayWorldScale = this.trayNode.worldScale;
    const destinationScale = new Vec3(
      80 / TILE_WIDTH * trayWorldScale.x / transitionWorldScale.x,
      78 / TILE_HEIGHT * trayWorldScale.y / transitionWorldScale.y,
      1,
    );

    // 脱离棋盘并保持屏幕位置不变；此后整段移动轨迹都处于界面最上层。
    node.setParent(this.foodTransitionLayer);
    node.setScale(startWorldScale.x / transitionWorldScale.x,
      startWorldScale.y / transitionWorldScale.y, 1);
    node.setPosition(transitionUI.convertToNodeSpaceAR(startWorld));
    node.setSiblingIndex(this.foodTransitionLayer.children.length - 1);
    const targetLocal = transitionUI.convertToNodeSpaceAR(targetWorld);
    const startScale = node.scale.clone();
    const movingLevel = this.levelData;
    tween(node)
      .to(0.08, {
        scale: new Vec3(startScale.x * 0.9, startScale.y * 0.9, 1),
      })
      .to(
        0.2,
        {
          position: targetLocal,
          scale: destinationScale,
        },
        {easing: 'cubicOut'},
      )
      .call(
        /**
         * 在动画到达指定阶段后更新游戏状态。
         */ () => {
          node.destroy();
          if (this.levelData !== movingLevel) {
            return;
          }
          this.tileNodes.delete(tile.id);
          this.refreshTileStates();
          this.resolveTray();
        },
      )
      .start();
    this.updateProgress();
  }

  /**
   * 将食材插入托盘中相同种类旁边。
   */
  protected insertIntoTray(type: IngredientType): number {
    let insertAt = this.tray.length;
    for (let i = this.tray.length - 1; i >= 0; i -= 1) {
      if (this.tray[i] === type) {
        insertAt = i + 1;
        break;
      }
    }
    this.tray.splice(insertAt, 0, type);
    return insertAt;
  }

  /**
   * 检查托盘中的连续三张并触发消除或失败。
   */
  protected resolveTray(): void {
    this.renderTray();
    const matchStart = this.findAdjacentMatch();
    if (matchStart >= 0) {
      this.combo += 1;
      this.animateMatch(matchStart);
    } else if (this.isBoardCleared()) {
      this.endGame(true);
    } else if (this.tray.length >= this.levelData.config.traySize) {
      this.endGame(false);
    } else {
      this.combo = 0;
      this.locked = false;
      this.checkpointGame();
    }
  }

  /**
   * 返回第一组三个连续同类食材的起始下标；没有则返回 -1。
   */
  protected findAdjacentMatch(): number {
    for (let i = 0; i <= this.tray.length - 3; i += 1) {
      if (this.tray[i] === this.tray[i + 1]
        && this.tray[i] === this.tray[i + 2]) {
        return i;
      }
    }
    return -1;
  }

  /**
   * 播放匹配动画并整理托盘。
   */
  protected abstract animateMatch(matchStart: number): void;

  /**
   * 清除仍在运行的匹配特效。
   */
  protected abstract clearMatchEffects(): void;

  /**
   * 判断棋盘是否已经清空。
   */
  protected abstract isBoardCleared(): boolean;
}
````

````typescript
  /**
   * 将调试棋盘缩减为可直接消除的三张牌。
   */
  protected leaveThreeMatchingTiles(): void {
    if (!this.levelData
      || this.hasPlayerMenu() ||
      (this.locked && !this.gameEnded)) {
      return;
    }
    // 从原有牌中保留同类三张，必要时取回已移除的牌，维持存档的种类总数。
    const ordered = this.levelData.tiles.slice().sort(
      /**
       * 按指定顺序比较并排列条目。
       */ (a, b) => b.layer - a.layer,
    );
    const type =
      ordered.find(
        /**
         * 按条件筛选或判断条目：!tile.removed。
         */ (tile) => !tile.removed,
      )?.type || ordered[0]?.type;
    const kept = ordered
      .filter(
        /**
         * 按条件筛选或判断条目：tile.type === type。
         */ (tile) => tile.type === type,
      )
      .slice(0, 3);
    if (kept.length !== 3) {
      return;
    }
    const keptIds = new Set(
      kept.map(
        /**
         * 将当前条目转换为后续处理所需的值。
         */ (tile) => tile.id,
      ),
    );

      },
    );

    this.reserve = [];
````

````typescript
export function referenceArea():ReferenceArea {
````

````typescript
import {

} from './HotPotUi';

/**
 * HotPotResources：拆分后的后段场景行为。
 */
export abstract class HotPotResources extends HotPotState {
  /**
   * 加载背景音乐和音效并设置播放源。
   */
  protected setupAudio(): void {
    const bgmNode = new Node('BgmAudio');
    this.node.addChild(bgmNode);
    this.bgmSource = bgmNode.addComponent(AudioSource);
    this.bgmSource.loop = true;
    this.bgmSource.volume = BGM_VOLUME;

    const effectNode = new Node('EffectAudio');
    this.node.addChild(effectNode);
    this.effectSource = effectNode.addComponent(AudioSource);
    this.effectSource.volume = EFFECT_VOLUME;

    this.node.on(Node.EventType.TOUCH_START, this.startBgmIfReady, this);

    const audioAssets: Array<{
      /**
       * key 对应的状态或配置。
       */ key: GameAudio;
      /**
       * path 对应的状态或配置。
       */ path: string;
    }> = [
      {key: 'bgm', path: 'audio/bgm'},
      {key: 'match', path: 'audio/match'},
      {key: 'win', path: 'audio/win'},
      {key: 'lose', path: 'audio/lose'},
    ];
    audioAssets.forEach(
      /**
       * 依次处理集合中的每个条目。
       */ ({key, path}) => {
        resources.load(
          path,
          AudioClip,
          /**
           * 在资源加载完成后更新有效节点或处理失败。
           */ (error, clip) => {
            if (error || !clip) {
              console.error(`[HotPotGame] 无法加载音频：${path}`, error);
              return;
            }
            this.audioClips.set(key, clip);
            if (key === 'bgm') {
              this.bgmSource.clip = clip;
            }
          },
        );
      },
    );
  }

  /**
   * 在允许播放时启动背景音乐。
   */
  protected startBgmIfReady(): void {
    if (!this.soundEnabled || !this.bgmSource?.clip || this.bgmSource.playing) {
      return;
    }
    this.bgmSource.play();
  }

  /**
   * 切换声音状态并更新按钮显示。
   */
  protected toggleSound(): void {
    this.soundEnabled = !this.soundEnabled;
    this.soundButtonLabel.string = this.soundEnabled ? '声音：开' : '声音：关';
    const soundIcon =
      this.soundButtonLabel.node.parent?.getChildByName('ControlIcon');
    if (soundIcon) {
      drawControlIcon(soundIcon, false, this.soundEnabled);
    }
    this.effectSource.volume = this.soundEnabled ? EFFECT_VOLUME : 0;
    if (this.soundEnabled) {
      this.startBgmIfReady();
    } else {
      this.bgmSource.pause();
      this.effectSource.stop();
    }
  }

  /**
   * 按指定音量播放游戏音效。
   */
  protected playEffect(key: Exclude<GameAudio, 'bgm'>, volume: number): void {
    if (!this.soundEnabled) {
      return;
    }
    const clip = this.audioClips.get(key);
    if (clip) {
      // 使用可停止的音源，关闭声音时也能立即停止当前音效。
      this.effectSource.stop();
      this.effectSource.clip = clip;
      this.effectSource.volume = volume * EFFECT_VOLUME;
      this.effectSource.play();
    }
  }

  /**
   * Both transparent sheets are shared by every theme. Run completion once,
   * after all 50 frame rectangles have been created.
   */
  protected loadIngredientImages(done: () => void): void {
    let pending = INGREDIENT_ART.length;
    INGREDIENT_ART.forEach((types, sheet) => {
      const path = 'ingredients/food-' + (sheet ? 'b' : 'a') + '/texture';
      resources.load(path, Texture2D, (error, texture) => {
        if (!this.node.isValid) return;
        if (!error && texture && texture.width >= 5 && texture.height >= 5
          && texture.width % 5 === 0 && texture.height % 5 === 0) {
          types.forEach((type, index) => {
            const frame = new SpriteFrame();
            frame.texture = texture;
            frame.rect = new Rect(index % 5 * texture.width / 5,
              Math.floor(index / 5) * texture.height / 5,
              texture.width / 5, texture.height / 5);
            this.ingredientFrames.set(type, frame);
          });
        } else {
          console.error('[HotPot] 食材资源不可用，保留文字牌面', path, error);
        }
        pending--;
        if (!pending) loadReferenceTileSkins(done);
      });
    });
  }
}
````

````typescript
function rgb(value: string): number[] {
  return [1, 3, 5].map((offset) => parseInt(value.slice(offset, offset + 2), 16));
}

/** Build both tile faces once; no canvas or per-tile Graphics buffers. */
export function createTilePixels(): Uint8Array {
  const theme = getPageTheme();
  const radius = theme.id === 4 ? 16 : theme.id === 9 ? 8 : 12;
  const pixels = new Uint8Array(CELL_WIDTH * 2 * CELL_HEIGHT * 4);
  for (let state = 0; state < 2; state++) {
    const colors = [[75, 43, 23, 28],
      [...rgb(theme.tileSide), 255], [...rgb(theme.tileEdge), 255],
      [...rgb(theme.tileFace), 255]].map((color, layer) =>
        state && layer ? color.map((v, i) => i === 3 ? v : Math.round(v * .52)) : color);
    for (let y = 0; y < CELL_HEIGHT; y++) {
      for (let x = 0; x < CELL_WIDTH; x++) {
        const result = [0, 0, 0, 0];
        for (const sx of [0.25, 0.75]) {
          for (const sy of [0.25, 0.75]) {
            const px = (x + sx) / CELL_WIDTH * 110.4 - 55.2;
            const py = 47.6 - (y + sy) / CELL_HEIGHT * 101.2;
            const sample = [0, 0, 0, 0];
            for (let layer = 0; layer < 4; layer++) {
              const inset = [1, 0, 0, 1.5][layer];
              const offset = [-7, -4, 0, 1][layer];
              const dx = Math.max(0, Math.abs(px) - (55.2 - radius - inset));
              const dy = Math.max(0,
                Math.abs(py - offset) - (45.6 - radius - inset));
              if (dx * dx + dy * dy > radius * radius) {
                continue;
              }
              const color = colors[layer];
              const alpha = color[3] / 255;
              for (let c = 0; c < 3; c++) {
                sample[c] = color[c] * alpha + sample[c] * (1 - alpha);
              }
              sample[3] = alpha + sample[3] * (1 - alpha);
            }
            if (Math.abs(py - 43.6) < 0.5 && Math.abs(px) < 42.2) {
              const highlight = state ? [140, 139, 133] : [255, 252, 242];
              sample.splice(0, 4, ...highlight, 1);
            }
            for (let c = 0; c < 4; c++) {
              result[c] += sample[c] / 4;
            }
          }
        }
        const at = (y * CELL_WIDTH * 2 + state * CELL_WIDTH + x) * 4;
        for (let c = 0; c < 3; c++) {
          pixels[at + c] = result[3] ? result[c] / result[3] : 0;
        }
        pixels[at + 3] = Math.round(result[3] * 255);
      }
    }
  }
  return pixels;
}
````

````typescript
      reward ||
        `炸掉 ${targets.length} 张${getIngredient(targets[0].type).name}`,
      '#A14425',
    );
    if (this.isBoardCleared()) {
      this.endGame(true);
    } else {
      this.checkpointGame();
    }
  }
}
````

````typescript
  width: number,
  height: number,
  tone: 'red' | 'cream' | 'muted' = 'red',
  emphasized = true,

  rememberSkin(node, {kind: 'button', width, height, tone, emphasized});
  const theme = getPageTheme();
  const body = tone === 'red' ? theme.accent : tone === 'muted' ? theme.disabled : theme.card;
  const radius = Math.min(theme.radius * .6, height / 2);
  const g = drawRoundRect(node, width, height, radius, theme.dark);

    const from = hex(theme.light), to = hex(theme.accent);
    const innerHeight = height - 14;
    const round = Math.max(4, radius - 4);
    for (let i = 0; i < 32; i++) {
      const t = i / 31;
      const highlight = (1 - t) * .16;
      g.fillColor = new Color(
        Math.round(from.r * highlight + to.r * (1 - highlight)),
        Math.round(from.g * highlight + to.g * (1 - highlight)),
        Math.round(from.b * highlight + to.b * (1 - highlight)), 255);
      const y = height / 2 - 5 - (i + 1) * innerHeight / 32;
      const distance = Math.max(0, Math.abs((i + .5) * innerHeight / 32
        - innerHeight / 2) - (innerHeight / 2 - round));
      const inset = round - Math.sqrt(Math.max(0, round * round - distance * distance));
      g.rect(-width / 2 + 5 + inset, y, width - 10 - inset * 2,
        innerHeight / 32 + .4);
      g.fill();
    }
  }
  g.strokeColor = hex(emphasized

  width: number,
  height: number,
  fill: string,
  callback: () => void,
): Node {
  const node = new Node(`Button_${text}`);

  node.setPosition(position);
  const primary =
    ['#D75542', '#C94E35', '#C84730', '#ED4A32', '#875139', '#A3613B'].indexOf(
      fill,
    ) >= 0;

    drawLacquerButton(node, width, height, primary ? 'red' : 'cream');

    drawRoundRect(node, width, height, 16, fill, '#D5B47D', 3);

  const label = makeLabel(
    node,
    text,
    width < 140 ? 24 : 26,
    primary ? getPageTheme().onAccent : getPageTheme().ink,
    new Vec3(0, 1),
    width - (width < 80 ? 16 : width < 140 ? 36 : 56),
    height - (width < 80 ? 14 : height <= 60 ? 20 : 28),
  );
  label.isBold = true;
  const homePosition = new Vec3(position.x, position.y, position.z);
  node.on(
    Node.EventType.TOUCH_START,
    /**
     * 响应节点触摸或游戏事件。
     */ () => {
      if ((node as PageSkinNode).pageDisabled) return;
      tween(node).stop();
      tween(node)
        .to(0.07, {scale: new Vec3(0.99, 0.99, 1),
          position: new Vec3(homePosition.x, homePosition.y - 4, homePosition.z)})
        .start();
    },
  );
  node.on(
    Node.EventType.TOUCH_END,
    /**
     * 响应节点触摸或游戏事件。
     */ () => {
      if ((node as PageSkinNode).pageDisabled) return;
      tween(node).stop();
      tween(node).to(0.08, {scale: Vec3.ONE, position: homePosition}).start();
      callback();
    },
  );
  node.on(
    Node.EventType.TOUCH_CANCEL,
    /**
     * 响应节点触摸或游戏事件。
     */ () => {
      tween(node).stop();
      tween(node).to(0.08, {scale: Vec3.ONE, position: homePosition}).start();
    },
  );
  return node;
}

  const skin = (node as PageSkinNode).pageSkin;
  if (skin?.kind === 'button' && !enabled) {

}
````

````typescript
      const remote = this.parseRemote(loaded);
      // 确认服务器身份之后才读取该账号的本地缓存，不展示上一账号的私有进度。
      this.playerId = remote.playerId;
      const cached = this.readLocal(remote.playerId);
      const browserGuest = this.cloud.platform === 'web' && !cached
        && remote.revision === 0 ? this.local.state : null;
      this.local =
        cached || this.blank(remote.playerId, remote.revision,
          browserGuest || remote.state);
      if (browserGuest) {
        this.local.dirty = true;
        this.local.mutationId = this.newMutationId();
        this.status = 'pending';
      } else if (cached?.dirty) {
        if (cached.mutationId === remote.lastMutationId) {
          this.local = this.blank(
            remote.playerId,
            remote.revision,
            remote.state,
          );
          this.status = 'synced';
        } else if (cached.baseRevision !== remote.revision) {
          this.remote = remote;
          this.status = 'conflict';
        } else {
          this.status = 'pending';
        }
      } else {
        this.local = this.blank(remote.playerId, remote.revision, remote.state);
        this.status = 'synced';
      }
      this.applyActiveDay();
      this.persist();
      this.changed();
      if (this.cloud.platform === 'web') {
        this.claimGuestWins();
      }
      void this.refreshLeaderboard();
      if (this.status === 'pending') {
        this.schedule();
      }
    } catch (error) {
      this.describeFailure(error);
      this.cloud = null;
      this.status = 'error';
      this.changed();
    }
  }

  /**
   * 跨日时清空当日对局并保留永久进度。
   */
  protected applyActiveDay(): boolean {
    if (!this.activePlayDate
      || this.local.state.playDate === this.activePlayDate) {
      return false;
    }
    this.local.state = enterPlayDay(this.local.state, this.activePlayDate);
    this.local.dirty = true;
    this.local.mutationId = this.newMutationId();
    if (this.status !== 'conflict') {
      this.status = this.playerId
        ? 'pending'
        : this.status === 'error'
          ? 'error'
          : 'local';
    }
    return true;
  }

  /**
   * 校验并持久化玩家状态，再安排云端同步。
   */
  protected setState(state: PlayerState): void {
    const valid = validatePlayerState(state, this.regions);
    if (JSON.stringify(valid) === JSON.stringify(this.local.state)) {
      return;
    }
    this.local.state = valid;
    this.local.dirty = true;
    this.local.mutationId = this.newMutationId();
    if (this.status !== 'conflict') {
      this.status = this.playerId
        ? 'pending'
        : this.status === 'error'
          ? 'error'
          : 'local';
    }
    this.persist();
    this.changed();
    this.schedule();
  }

  /**
   * 按版本号提交本机修改并处理云端冲突。
   */
  protected async drain(): Promise<void> {
    if (!this.cloud
      || !this.playerId ||
      this.status === 'conflict') {
      return;
    }
    await this.flushRegionWins();
    while (this.local.dirty && !this.isConflict) {
      const sent = clone(this.local);
      this.status = 'syncing';
      this.changed();
      try {
        const response = await this.request({
          action: 'save',
          expectedRevision: sent.baseRevision,
          mutationId: sent.mutationId,
          state: sent.state,
        });
        const remote = this.parseRemote(response);
        if (remote.playerId !== this.playerId) {
          throw new Error('云端身份不一致');
        }
        if (response.conflict) {
          this.remote = remote;
          this.status = 'conflict';
          this.changed();
          return;
        }
        this.local.baseRevision = remote.revision;
        if (this.local.mutationId === sent.mutationId) {
          this.local.dirty = false;
          this.local.state = remote.state;
          this.status = 'synced';
        } else {
          // 云端可能合并了另一设备的永久记录，保留同步期间新操作及云端图鉴。
          this.local.state.progress = mergeProgress(
            this.local.state.progress,
            remote.state.progress,
          );
          this.status = 'pending';
        }
        this.applyActiveDay();
        this.persist();
        this.changed();
      } catch (error) {
        this.describeFailure(error);
        this.status = 'error';
        this.changed();
        return;
      }
    }
  }

  /**
   * 延迟提交玩家状态，合并短时间内的连续操作。
   */
  protected schedule(): void {
    if (!this.cloud
      || !this.playerId ||
      this.timer ||
      this.status === 'conflict') {
      return;
    }
    this.timer = setTimeout(
      /**
       * 到期后继续同步或完成动画。
       */ () => {
        this.timer = null;
        void this.flush();
      },
      1500,
    );
  }

  /**
   * 从缓存、云存储或云函数加载地区树。
   */
  protected async loadRegions(): Promise<void> {
    const key =
      'hotpot.regions.dev.v1:' +
      String(this.config.environmentId) +
      ':' +
      String(this.config.regionsFileId || 'bundled') +
      '';
    try {
      const cached = this.storage.getItem(key);
      if (cached) {
        this.regions = validateRegions(JSON.parse(cached));
      }
    } catch {
      /*
       * 损坏缓存使用随包数据。
       */
    }
    if (!this.config.regionsFileId) {
      return;
    }
    try {
      const downloaded: any = await this.callbackRequest(
        /**
         * 在当前操作完成后更新相关游戏状态。
         */ (success, fail) =>
          this.cloud.downloadFile({
            fileID: this.config.regionsFileId,
            success,
            fail,
          }),
      );
      const file: any = await this.callbackRequest(
        /**
         * 在当前操作完成后更新相关游戏状态。
         */ (success, fail) =>
          this.wxHost.getFileSystemManager().readFile({
            filePath: downloaded.tempFilePath,
            encoding: 'utf8',
            success,
            fail,
          }),
      );
      this.regions = validateRegions(JSON.parse(file.data));
      try {
        this.storage.setItem(key, JSON.stringify(this.regions));
      } catch {
        /*
         * 地区缓存失败不阻塞登录。
         */
      }
    } catch {
      // 旧地区文件可以继续保持私有权限；新增云函数只读该文件。
      try {
        const response = await this.request({action: 'regions'});
        this.regions = validateRegions(response.regions);
        try {
          this.storage.setItem(key, JSON.stringify(this.regions));
        } catch {
          /*
           * 缓存失败不阻塞游戏。
           */
        }
      } catch {
        /*
         * 首次离线仍可选择随包地区。
         */
      }
    }
  }

  /**
   * 调用玩家云函数并检查返回值与协议版本。
   */
  protected async request(data: any): Promise<any> {
    const response: any = await this.callbackRequest(
      /**
       * 在当前操作完成后更新相关游戏状态。
       */ (success, fail) =>
        this.cloud.callFunction({
          name: this.config.functionName,
          config: {env: this.config.environmentId},
          data,
          success,
          fail,
        }),
    );
    if (!response.result?.ok) {
      throw new Error(response.result?.code || '云端请求失败');
    }
    if (!['regions', 'config'].includes(data.action)
      && response.result.progressVersion !== PROGRESS_VERSION) {
      throw new Error('CLOUD_UPGRADE_REQUIRED');
    }
    if (!['regions', 'config'].includes(data.action)
      && response.result.playSessionVersion !== PLAY_SESSION_VERSION) {
      throw new Error('DAILY_UPGRADE_REQUIRED');
    }
    return response.result;
  }

  /**
   * 把云端异常归类为安全的用户提示。
   */
  protected describeFailure(error: any): void {
    const messages = [error?.message, error?.errMsg].filter(
      /**
       * 按条件筛选或判断条目：typeof value === 'string'。
       */ (value) => typeof value === 'string',
    );
    const message = messages.join(' ');
    if (/login_type_disabled|匿名登录/.test(message)) {
      this.failureText = '请在云环境开启网页访客登录';
      return;
    }
    if (/permission_denied|INVALID_ORIGIN|安全域名|安全来源/i.test(message)) {
      this.failureText = '请检查网页登录权限及安全来源配置';
      return;
    }
    const descriptions: Record<string, string> = {
      IDENTITY_REQUIRED: '微信身份未确认，请检查云环境关联',
      WEB_IDENTITY_REQUIRED: '网页登录未完成，请重试',
      WEB_SDK_UNAVAILABLE: '网页云服务加载失败，请检查网络后重试',
      DEVELOPMENT_ENV_REQUIRED: '云环境配置不匹配',
      DATABASE_UNAVAILABLE: '云存档服务暂不可用',
      INVALID_STATE: '当前存档校验失败',
      CLOUD_UPGRADE_REQUIRED: '请先更新云函数再同步图鉴与成就',
      DAILY_UPGRADE_REQUIRED:
        '请先更新云函数再同步每日关卡、随机牌阵与食材组合',
    };
    // 微信有时只在外层给出 -1，具体平台错误码包含在 errMsg 中。
    const embeddedCodes = (
      message.match(/(?:errCode|errorCode)\s*[:=]\s*-?\d+/gi) || []
    ).map(
      /**
       * 将当前条目转换为后续处理所需的值。
       */ (value) => value.replace(/^[\s\S]*[:=]\s*/, ''),
    );
    const platformCode = embeddedCodes.find(
      /**
       * 按条件筛选或判断条目：Math.abs(Number(value)) > 1。
       */ (value) => Math.abs(Number(value)) > 1,
    );
    const code =
      Object.keys(descriptions).find(
        /**
         * 按条件筛选或判断条目：messages.indexOf(item) >= 0。
         */ (item) => messages.indexOf(item) >= 0,
      ) ||
      (message.includes('超时')
        ? 'TIMEOUT'
        : platformCode ||
          String(error?.errCode || error?.code || 'NETWORK_ERROR'));
    this.failureText =
      descriptions[code] ||
      (code === 'TIMEOUT' ? '云端请求超时，请重试' : '云端连接失败，请重试');
    // 只记录已知业务代码或平台数字错误码，不输出响应、OpenID 或原始异常。
    const reason = [
      'INVALID_ENV',
      'FUNCTIONS_EXECUTE_FAIL',
      'FUNCTIONS_NOT_FOUND',
      'DATABASE_PERMISSION_DENIED',
      'FUNCTIONS_TIME_LIMIT_EXCEEDED',
      'RESOURCE_INSUFFICIENT',
    ].find(
      /**
       * 按条件筛选或判断条目：message.includes(value)。
       */ (value) => message.includes(value),
    );
    console.warn(
      '[HotPot Cloud Dev]',
      Object.hasOwnProperty.call(descriptions, code) ||
        /^-?\d+$/.test(code) ||
        code === 'TIMEOUT'
        ? code
        : 'NETWORK_ERROR',
      reason || '',
    );
  }

  /**
   * 将微信回调接口包装为带超时的 Promise。
   */
  protected callbackRequest(
    start: (success: (value: any) => void, fail: (error: any) => void) => void,
  ): Promise<any> {
    return new Promise(
      /**
       * 在当前操作完成后更新相关游戏状态。
       */ (resolve, reject) => {
        const timer = setTimeout(
          /**
           * 到期后继续同步或完成动画。
           */ () => reject(new Error('云端连接超时')),
          8000,
        );
        try {
          start(
            /**
             * 在当前操作完成后更新相关游戏状态。
             */ (value) => {
              clearTimeout(timer);
              resolve(value);
            },
            /**
             * 在当前操作完成后更新相关游戏状态。
             */ (error) => {
              clearTimeout(timer);
              reject(error);
            },
          );
        } catch (error) {
          clearTimeout(timer);
          reject(error);
        }
      },
    );
  }

  /**
   * 校验云端玩家档案的结构和状态。
   */
  protected parseRemote(response: any): CloudPlayer {
    const record = response.player;
    if (!record
      || !/^[a-f0-9]{32}$/.test(record.playerId) ||
      !Number.isSafeInteger(record.revision) ||
      record.revision < 0 ||
      !Number.isSafeInteger(record.updatedAt) ||
      typeof record.lastMutationId !== 'string') {
      throw new Error('云端档案格式错误');
    }
    return {...record, state: validatePlayerState(record.state, this.regions)};
  }

  /**
   * 生成按云环境和玩家身份隔离的本机缓存键。
   */
  protected key(owner: string | null): string {
    return (
      'hotpot.player.dev.v1:' +
      String(this.config.environmentId || 'unconfigured') +
      ':' +
      String(owner || 'guest') +
      ''
    );
  }

  /**
   * 创建指定玩家和云端版本的本机存档。
   */
  protected blank(
    owner: string | null,
    revision: number,
    state: PlayerState,
  ): LocalPlayer {
    return {
      version: 1,
      owner,
      baseRevision: revision,
      dirty: false,
      mutationId: '',
      state: clone(state),
    };
  }

  /**
   * 读取并验证当前玩家的本机缓存。
   */
  protected readLocal(owner: string | null): LocalPlayer | null {
    try {
      const value = JSON.parse(this.storage.getItem(this.key(owner)) || 'null');
      if (!value
        || value.version !== 1 ||
        value.owner !== owner ||
        !Number.isSafeInteger(value.baseRevision) ||
        value.baseRevision < 0 ||
        typeof value.dirty !== 'boolean' ||
        typeof value.mutationId !== 'string') {
        return null;
      }
      return {...value, state: validatePlayerState(value.state, this.regions)};
    } catch {
      return null;
    }
  }

  /**
   * 将当前玩家状态写入本机缓存。
   */
  protected persist(): void {
    try {
      this.storage.setItem(this.key(this.playerId), JSON.stringify(this.local));
      this.storageAvailable = true;
    } catch {
      this.storageAvailable = false;
    }
  }

  /**
   * 生成本次本机修改的幂等标识。
   */
  protected newMutationId(): string {
    return (
      '' +
      String(Date.now().toString(36)) +
      '-' +
      String((++this.mutationSequence).toString(36)) +
      '-' +
      String(Math.random().toString(36).slice(2, 14)) +
      ''
    );
  }

  /**
   * 通知界面刷新玩家状态。
   */
  protected changed(): void {
    this.onChange();
  }
}
````

````typescript
      remote.state.progress,
    );
    this.remote = null;
    if (useCloud) {
      this.local = this.blank(this.playerId, remote.revision, remote.state);
      this.local.state.progress = progress;
      this.local.dirty =
        JSON.stringify(progress) !== JSON.stringify(remote.state.progress);
      if (this.local.dirty) {
        this.local.mutationId = this.newMutationId();
      }
      this.status = this.local.dirty ? 'pending' : 'synced';
    } else {
      this.local.state.progress = progress;
      this.local.baseRevision = remote.revision;
      this.local.dirty = true;
      this.local.mutationId = this.newMutationId();
      this.status = 'pending';
    }
    this.applyActiveDay();
    this.persist();
    this.changed();
    this.schedule();
    return this.state;
  }
  /**
   * 连接云端账号。
   */
  protected abstract connect(): Promise<void>;
  /**
   * 根据当前自然日调整存档。
   */
  protected abstract applyActiveDay(): boolean;
  /**
   * 写入新的玩家档案。
   */
  protected abstract setState(state: PlayerState): void;
  /**
   * 发送等待同步的修改。
   */
  protected abstract drain(): Promise<void>;
  /**

   */
  protected abstract schedule(): void;
  /**

   */
  protected abstract loadRegions(): Promise<void>;

   */
  protected abstract request(data: any): Promise<any>;
  /**

  protected abstract describeFailure(error: any): void;
  /**
   * 包装平台回调请求。

  protected abstract callbackRequest(
    start: (success: (value: any) => void, fail: (error: any) => void) => void,
  ): Promise<any>;
  /**
   * 校验云端响应。
   */
  protected abstract parseRemote(response: any): CloudPlayer;
  /**
   * 创建本地存档键名。
   */
  protected abstract key(owner: string | null): string;
  /**

  protected abstract blank(
````

````typescript
import {GameSave, RegionNode, clone} from '../data/PlayerData';
import {PlayerServiceBase} from './PlayerServiceBase';

interface PendingWin {
  save: GameSave;
  regionCodes: string[];
  playDate: string;
}

/** 云榜缓存和离线通关队列独立于存档冲突选择。 */
export abstract class RegionLeaderboardService extends PlayerServiceBase {
  regionWins: Record<string, number> = {};
  leaderboardText = '尚未获取云端排行榜';
  private leaderboardRequest: Promise<void> | null = null;
  private winRequest: Promise<void> | null = null;

  /** 玩家主动同步时，将未登录期间的本机通关归到已确认的当前账号。 */
  claimGuestWins(): void {
    if (!this.playerId) {
      return;
    }
    const guestKey = `${this.key(null)}:pending-region-wins`;
    try {
      const guest = JSON.parse(this.storage.getItem(guestKey) || '[]');
      if (!Array.isArray(guest) || !guest.length) {
        return;
      }
      // 先写账号队列，后清访客队列；中途失败也不会丢记录，云端负责去重。
      this.storage.setItem(this.winKey(),
        JSON.stringify([...this.pendingWins(), ...guest]));
      this.storage.setItem(guestKey, '[]');
      this.schedule();
    } catch {
      this.leaderboardText = '本机通关待同步，请重试';
    }
  }

  regionCount(code: string): number {
    return this.regionWins[code] || 0;
  }

  sortedRegions(regions: RegionNode[]): RegionNode[] {
    return regions.map((region, index) => ({region, index}))
      .sort((a, b) => this.regionCount(b.region.code)
        - this.regionCount(a.region.code) || a.index - b.index)
      .map((item) => item.region);
  }

  setSave(save: GameSave): void {
    const previous = this.local.state.save;
    super.setSave(save);
    if (save.levelIndex !== 2 || save.status !== 'won'
      || previous?.status === 'won'
      || !this.local.state.profile.regionCodes.length) {
      return;
    }
    const pending = this.pendingWins();
    pending.push({
      save: clone(this.local.state.save!),
      regionCodes: this.local.state.profile.regionCodes.slice(),
      playDate: this.local.state.playDate,
    });
    this.writePendingWins(pending);
    this.schedule();
  }

  /** 失败保留队列；服务器按玩家和牌局去重。 */
  protected flushRegionWins(): Promise<void> {
    if (!this.winRequest) {
      this.winRequest = this.sendRegionWins().finally(() => {
        this.winRequest = null;
      });
    }
    return this.winRequest;
  }

  private async sendRegionWins(): Promise<void> {
    const owner = this.playerId;
    const pending = this.pendingWins();
    while (pending.length && this.cloud && owner === this.playerId) {
      try {
        await this.request({action: 'regionWin', win: pending[0]});
        pending.shift();
        // 合并请求期间产生的新通关，避免被旧队列覆盖。
        const current = this.pendingWins();
        current.shift();
        this.writePendingWins(current);
      } catch (error) {
        this.leaderboardText = this.rankingFailure(error);
        return;
      }
    }
  }

  refreshLeaderboard(): Promise<void> {
    if (this.leaderboardRequest) {
      return this.leaderboardRequest;
    }
    this.leaderboardRequest = this.fetchLeaderboard().finally(() => {
      this.leaderboardRequest = null;
    });
    return this.leaderboardRequest;
  }

  private async fetchLeaderboard(): Promise<void> {
    const key = `hotpot.region-ranking:${this.config.environmentId}`;
    try {
      const cached = JSON.parse(this.storage.getItem(key) || 'null');
      if (cached) {
        this.regionWins = this.validateCounts(cached);
        this.leaderboardText = '上次云端排行 · 当前离线';
      }
    } catch {
      // 损坏的展示缓存不阻止获取最新云榜。
    }
    if (!this.cloud || !this.playerId) {
      this.leaderboardText = '云端未连接 · 通关保留在本机等待上传';
      return;
    }
    try {
      await this.flushRegionWins();
      const response = await this.request({action: 'leaderboard'});
      this.regionWins = this.validateCounts(response.counts);
      this.leaderboardText = this.pendingWins().length
        ? '云端排行 · 本机仍有通关待同步'
        : '云端排行 · 完成第三关计 1 次';
      try {
        this.storage.setItem(key, JSON.stringify(this.regionWins));
      } catch {
        // 云榜缓存失败不影响这次展示或玩家存档。
      }
    } catch (error) {
      this.leaderboardText = this.rankingFailure(error);
    }
  }

  private rankingFailure(error: any): string {
    const message = String(error?.message || '');
    if (/INVALID_ACTION|UPGRADE_REQUIRED/.test(message)) {
      return '云榜服务尚未更新 · 本机通关已保留';
    }
    if (message === 'DATABASE_UNAVAILABLE') {
      return '云榜数据库暂不可用 · 本机通关已保留';
    }
    return '云榜连接失败 · 通关记录待联网补传';
  }

  private validateCounts(value: any): Record<string, number> {
    if (!value || typeof value !== 'object' || Array.isArray(value)) {
      throw new Error('排行榜格式错误');
    }
    const result: Record<string, number> = {};
    const visit = (nodes: RegionNode[]): void => {
      for (const node of nodes) {
        const count = value[node.code];
        if (Number.isSafeInteger(count) && count >= 0) {
          result[node.code] = count;
        }
        visit(node.children || []);
      }
    };
    visit(this.regions);
    return result;
  }

  private pendingWins(): PendingWin[] {
    try {
      const value = JSON.parse(this.storage.getItem(this.winKey()) || '[]');
      return Array.isArray(value) ? value : [];
    } catch {
      return [];
    }
  }

  private writePendingWins(wins: PendingWin[]): void {
    try {
      this.storage.setItem(this.winKey(), JSON.stringify(wins));
    } catch {
      this.storageAvailable = false;
    }
  }

  private winKey(): string {
    return `${this.key(this.playerId)}:pending-region-wins`;
  }
}
````

````typescript
/** 浏览器按需加载官方 SDK；微信端继续使用内置云接口，不打包此 SDK。 */
const SDK_URL =
  'https://static.cloudbase.net/cloudbase-js-sdk/3.10.4/cloudbase.full.js';
let sdkPromise: Promise<any> | null = null;

export function hasWebCloudRuntime(): boolean {
  return !!(globalThis as any).document?.createElement;
}

function loadSdk(): Promise<any> {
  const browser = globalThis as any;
  if (browser.cloudbase?.init) {
    return Promise.resolve(browser.cloudbase);
  }
  if (!sdkPromise) {
    sdkPromise = new Promise((resolve, reject) => {
      const script = browser.document.createElement('script');
      const fail = (): void => {
        clearTimeout(timer);
        script.remove();
        sdkPromise = null;
        reject(new Error('WEB_SDK_UNAVAILABLE'));
      };
      const timer = setTimeout(fail, 15000);
      script.src = SDK_URL;
      script.async = true;
      script.onerror = fail;
      script.onload = (): void => {
        clearTimeout(timer);
        if (!browser.cloudbase?.init) {
          fail();
          return;
        }
        resolve(browser.cloudbase);
      };
      browser.document.head.appendChild(script);
    });
  }
  return sdkPromise;
}

/** SDK 负责持久会话与令牌刷新；不将调用者提供的身份当作云端身份。 */
export async function createWebCloud(config: {
  environmentId: string;
}): Promise<any> {
  const sdk = await loadSdk();
  const app = sdk.init({env: config.environmentId});
  const auth = app.auth({persistence: 'local'});
  let session = await auth.getSession();
  // 无效/过期凭据需要重新访客登录；网络故障则保留身份等待重试。
  // 只更新 SDK 会话，不清理游戏存档或待上传通关记录。
  if (session.error && session.error.category !== 'INVALID_CREDENTIALS') {
    throw session.error;
  }
  if (!session.data?.session) {
    session = await auth.signInAnonymously();
    if (session.error) {
      throw session.error;
    }
  }
  if (!session.data?.session) {
    throw new Error('WEB_IDENTITY_REQUIRED');
  }
  return {
    platform: 'web',
    init(): void {},
    callFunction({name, data, success, fail}: any): void {
      app.callFunction({name, data}).then(success, fail);
    },
  };
}
````
