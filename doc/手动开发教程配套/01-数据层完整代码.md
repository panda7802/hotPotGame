/**
 * 检查云开发环境、地区文件和资源名称是否符合开发配置。
 */
export function validateDevelopmentConfig(config = CLOUD_CONFIG): void {
  if (!config.enabled) {
    throw new Error('开发云环境尚未配置');
  }
  if (!config.environmentId
    || (config.environmentId === RELEASE_ENVIRONMENT_ID
      && !config.allowSharedEnvironment)) {
    throw new Error('必须使用独立开发云环境，或明确授权只新增 Dev 资源');
  }
  if (config.regionsFileId
    && !config.regionsFileId.startsWith(`cloud://${config.environmentId}.`)) {
    throw new Error('地区文件必须来自开发云环境');
  }
  if (config.functionName !== 'hotpotPlayerDev'
    || config.collectionName !== 'hotpot_players_dev') {
    throw new Error('开发云函数或集合名称错误');
  }
}

/**
 * AchievementEntry 的数据结构；字段说明见各成员。
 */
export interface AchievementEntry {
  /**
   * 实体的唯一编号。
   */
  id: string;
  /**
   * 界面显示的标题。
   */ title: string;
  /**
   * 成就的说明文字。
   */ description: string;
  /**
   * 当前完成数值。
   */ current: number;
  /**
   * 解锁所需的目标数值。
   */ target: number;
  /**
   * 成就是否已经解锁。
   */ unlocked: boolean;
}

/**
 * 根据永久进度计算各成就的目标和解锁状态。
 */
export function achievementEntries(state: PlayerState): AchievementEntry[] {
  const p = state.progress;
  const best = Math.max(...p.bestCleared);
  const result: AchievementEntry[] = [];
  const add =
    /**
     * 处理 add 对应的游戏逻辑。
     */ (
      id: string,
      title: string,
      description: string,
      current: number,
      target: number,
    ): void => {
      result.push({
        id,
        title,
        description,
        current: Math.min(current, target),
        target,
        unlocked: current >= target,
      });
    };
  add(
    'first-taste',
    '开锅尝鲜',
    '拿取食材，点亮第一份图鉴',
    p.discovered.length,
    1,
  );
  add('first-clear', '初次开席', '在一局中消除至少3份食材', best, 3);
  add(
    'hometown',
    '家乡有味',
    '为小馆选择自己的地区',
    p.regionChosen ? 1 : 0,
    1,
  );
  add('five-flavors', '五味俱全', '发现5种不同食材', p.discovered.length, 5);
  add('ten-flavors', '十全食客', '发现10种不同食材', p.discovered.length, 10);
  add(
    'full-menu',
    '满汉小馆',
    `点亮全部${INGREDIENTS.length}种食材图鉴`,
    p.discovered.length,
    INGREDIENTS.length,
  );
  add('clear-30', '热锅熟手', '在一局中消除至少30份食材', best, 30);
  add('clear-100', '百味入锅', '在一局中消除至少100份食材', best, 100);
  add('clear-300', '火候大师', '在一局中消除至少300份食材', best, 300);
  add(
    'win-1',
    '第一锅香',
    '通关第1关',
    p.completedLevels.indexOf(0) >= 0 ? 1 : 0,
    1,
  );
  add(
    'win-2',
    '热锅行家',
    '通关第2关',
    p.completedLevels.indexOf(1) >= 0 ? 1 : 0,
    1,
  );
  add(
    'win-3',
    '千层掌勺',
    '通关第3关',
    p.completedLevels.indexOf(2) >= 0 ? 1 : 0,
    1,
  );
  add(
    'clean-win',
    '一气呵成',
    '不使用复活通关任意一关',
    p.cleanWins.length ? 1 : 0,
    1,
  );
  add('all-wins', '三锅圆满', '完成全部3关', p.completedLevels.length, 3);
  return result;
}

/**
 * 返回已经解锁的成就编号。
 */
export function unlockedAchievementIds(state: PlayerState): string[] {
  return achievementEntries(state)
    .filter(
      /**
       * 按条件筛选或判断条目：item.unlocked。
       */ (item) => item.unlocked,
    )
    .map(
      /**
       * 将当前条目转换为后续处理所需的值。
       */ (item) => item.id,
    );
}

/**
 * 由整数种子生成可重现的伪随机数。
 */
function seededRandom(seed: number): () => number {
  let value = seed % 2147483647;
  if (value <= 0) {
    value += 2147483646;
  }
  /**
   * 在当前操作完成后更新相关游戏状态。
   */
  return () => {
    value = (value * 16807) % 2147483647;
    return (value - 1) / 2147483646;
  };
}

/**
 * 使用传入随机源打乱数组。
 */
function shuffle<T>(list: T[], random: () => number): T[] {
  for (let i = list.length - 1; i > 0; i -= 1) {
    const j = Math.floor(random() * (i + 1));
    [list[i], list[j]] = [list[j], list[i]];
  }
  return list;
}

/**
 * 按食材池生成可成组三张的牌型序列。
 */
function buildTypes(
  count: number,
  typeCount: number,
  typeOffset: number,
  layerIndex: number,
  random: () => number,
  pool?: IngredientType[],
): IngredientType[] {
  const availableTypeCount = Math.max(
    1,
    Math.min(10, typeCount, INGREDIENTS.length),
  );
  const groupTypes: number[] = [];
  for (let i = 0; i < count / 3; i += 1)
    groupTypes.push((i + layerIndex) % availableTypeCount);
  shuffle(groupTypes, random);
  const result: IngredientType[] = [];
  /**
   * 依次处理集合中的每个条目。
   */
  groupTypes.forEach((typeIndex) => {
    const ingredient = pool
      ? pool[typeIndex]
      : INGREDIENTS[(typeIndex + typeOffset) % 13].id;
    result.push(ingredient, ingredient, ingredient);
  });
  return shuffle(result, random);
}

export const EIGHT_ROW_IMAGE_HEIGHT = 68 * 1.2 * 1.2 * 1.15 * 0.82;
const EIGHT_ROW_SPACING = Math.ceil(EIGHT_ROW_IMAGE_HEIGHT * 1.1);

/**
 * 计算一层棋盘的牌位坐标。
 */
function buildPositions(
  count: number,
  layerIndex: number,
  targetRows?: number,
): Array<{
  /**
   * 棋盘横坐标。
   */
  x: number;
  /**
   * 棋盘纵坐标。
   */
  y: number;
}> {
  const columns = targetRows
    ? Math.ceil(count / targetRows)
    : count <= 9
      ? 3
      : count <= 15
        ? 5
        : 6;
  const rows = Math.ceil(count / columns);
  const spacingX = targetRows
    ? columns <= 3
      ? 176
      : 156
    : columns === 6
      ? 112
      : 120;
  const spacingY = targetRows ? EIGHT_ROW_SPACING : 108;
  const offsets = [
    {x: -27, y: -18},
    {x: 24, y: 17},
    {x: -11, y: 39},
    {x: 38, y: -2},
    {x: 0, y: 0},
  ];
  const baseOffset = offsets[layerIndex % offsets.length];
  // 八行牌阵每层沿不同方向错开，让下层露出牌角与侧边。
  // 纵向偏移限制在 16px 内，为最上/下排保留棋盘边距。
  const stackedOffsets = [
    {x: -28, y: -16},
    {x: 0, y: 0},
    {x: 28, y: 16},
    {x: -12, y: 8},
  ];
  const offset = targetRows
    ? stackedOffsets[layerIndex % stackedOffsets.length]
    : baseOffset;
  /**
   * 棋盘横坐标。
   */
  /**
   * 棋盘纵坐标。
   */
  const result: Array<{x: number; y: number}> = [];
  for (let i = 0; i < count; i += 1) {
    const row = Math.floor(i / columns);
    const itemsInRow = Math.min(columns, count - row * columns);
    const col = i - row * columns;
    // 相邻行左右交错，不改变同层行距，也不依赖随机数。
    const rowStagger = targetRows ? (row % 2 === 0 ? -10 : 10) : 0;
    result.push({
      x: (col - (itemsInRow - 1) / 2) * spacingX + offset.x + rowStagger,
      y: ((rows - 1) / 2 - row) * spacingY + offset.y,
    });
  }
  return result;
}

// Six separate groups alternate square/diamond arrangements. Every tile
// overlaps two tiles in the adjacent layer by at least 20% of its area,
// without overlaps within a layer or cumulative drift across 37 layers.
/**
 * 生成第三关的原有分层布局。
 */
function buildLevelThreePositions(
  config: LevelConfig,
  layoutSeed: number,
): Array<
  Array<{
    /**
     * 棋盘横坐标。
     */
    x: number;
    /**
     * 棋盘纵坐标。
     */
    y: number;
  }>
> {
  const random = seededRandom(layoutSeed);
  // Seed zero keeps the original geometry for saves made before random layouts.
  // Group centers stay fixed through all layers, so randomization cannot create
  // drifting stacks or small accidental overlaps between neighboring groups.
  const randomized = layoutSeed !== 0;
  const spacingX = randomized ? 276 + Math.floor(random() * 35) : 288;
  const spacingY = randomized ? 260 + Math.floor(random() * 19) : 252;
  const halfWidth = randomized ? 46 : 48;
  const halfHeight = randomized ? 38 : 42;
  /**
   * 将当前条目转换为后续处理所需的值。
   */
  const rowShifts = [0, 1, 2].map(() =>
    randomized ? Math.floor(random() * 71) - 35 : 0,
  );
  /**
   * 将当前条目转换为后续处理所需的值。
   */
  const columnShifts = [0, 1].map(() =>
    randomized ? Math.floor(random() * 17) - 8 : 0,
  );
  /**
   * 在当前操作完成后更新相关游戏状态。
   */
  const phases = Array.from({length: 6}, () =>
    randomized ? Math.floor(random() * 2) : 0,
  );
  /**
   * 将当前条目转换为后续处理所需的值。
   */
  return config.layers.map((_, layer) => {
    /**
     * 棋盘横坐标。
     */
    /**
     * 棋盘纵坐标。
     */
    const positions: Array<{x: number; y: number}> = [];
    for (let row = 0; row < 3; row += 1) {
      for (let col = 0; col < 2; col += 1) {
        const offsets =
          (layer + phases[row * 2 + col]) % 2 === 0
            ? [
                {x: -halfWidth, y: -halfHeight},
                {x: halfWidth, y: -halfHeight},
                {x: -halfWidth, y: halfHeight},
                {x: halfWidth, y: halfHeight},
              ]
            : [
                {x: -halfWidth * 2, y: 0},
                {x: halfWidth * 2, y: 0},
                {x: 0, y: -halfHeight * 2},
                {x: 0, y: halfHeight * 2},
              ];
        /**
         * 依次处理集合中的每个条目。
         */
        offsets.forEach((offset) =>
          positions.push({
            x: (col - 0.5) * spacingX + rowShifts[row] + offset.x,
            y: (1 - row) * spacingY + columnShifts[col] + offset.y,
          }),
        );
      }
    }
    return positions;
  });
}

// Keep the old seed range reproducible for existing saves. New seeds use
// visibly
// different formations and randomized ingredient placement, not small offsets.
export const RANDOM_LAYOUT_SEED_START = 2147483647;
export const RANDOM_LAYOUT_SEED_END = 4294967292;
export const INGREDIENT_SEED_MAX = 2147483646;

/**
 * 新局抽取整个食材组合；牛羊肉互斥，翻锅只重排该组合已有的牌。
 */
export function selectLevelIngredients(
  levelIndex: number,
  seed: number,
  poolVersion: 1 | 2 = 2,
): IngredientType[] {
  const config = LEVELS[levelIndex];
  /**
   * 在当前操作完成后更新相关游戏状态。
   */
  if (seed === 0) {
    return Array.from(
      {length: config.typeCount},
      (_, i) => INGREDIENTS[(i + (config.typeOffset || 0)) % 13].id,
    );
  }
  /**
   * 将当前条目转换为后续处理所需的值。
   */
  const shuffled = shuffle(
    (poolVersion === 1 ? INGREDIENTS.slice(0, 23) : INGREDIENTS)
      .map((item) => item.id),
    seededRandom(seed),
  );
  const selected: IngredientType[] = [];
  for (const type of shuffled) {
    if ((type === 'beef' && selected.indexOf('lamb') >= 0)
      || (type === 'lamb' && selected.indexOf('beef') >= 0)) {
      continue;
    }
    selected.push(type);
    if (selected.length === config.typeCount) {
      break;
    }
  }
  return selected;
}

/**
 * 根据种子选取第三关的布局族。
 */
export function levelThreeLayoutFamily(seed = 0): number {
  return seed < RANDOM_LAYOUT_SEED_START
    ? -1
    : (seed - RANDOM_LAYOUT_SEED_START) % 3;
}

/**
 * 根据种子生成第三关的随机布局。
 */
function buildNewLevelThreePositions(
  config: LevelConfig,
  seed: number,
): Array<
  Array<{
    /**
     * 棋盘横坐标。
     */
    x: number;
    /**
     * 棋盘纵坐标。
     */
    y: number;
  }>
> {
  const family = levelThreeLayoutFamily(seed);
  const centers =
    family === 0
      ? [
          {x: -172, y: 235},
          {x: 172, y: 235},
          {x: -172, y: -235},
          {
            x: 172,
            y: -235,
          },
        ]
      : family === 1
        ? [
            {x: 0, y: 206},
            {x: 0, y: -206},
          ]
        : [
            {x: 0, y: 280},
            {x: -180, y: 0},
            {x: 180, y: 0},
            {x: 0, y: -280},
          ];
  const count = family === 1 ? 12 : 6;
  const radiusX = family === 1 ? 2.76 : 1.18;
  const radiusY = family === 1 ? 2.24 : 1.18;
  const width = 92 * 1.2 * config.tileScale;
  const height = 76 * 1.2 * config.tileScale;
  const random = seededRandom(seed);
  /**
   * 将当前条目转换为后续处理所需的值。
   */
  const phases = centers.map(() => Math.floor(random() * 2));
  // Alternating half-step rings cover every lower tile with at least two
  // upper tiles. Their tested radii prevent same-layer and incidental overlaps.
  /**
   * 将当前条目转换为后续处理所需的值。
   */
  return config.layers.map((_, layer) => {
    /**
     * 棋盘横坐标。
     */
    /**
     * 棋盘纵坐标。
     */
    const positions: Array<{x: number; y: number}> = [];
    /**
     * 依次处理集合中的每个条目。
     */
    centers.forEach((center, group) => {
      for (let i = 0; i < count; i += 1) {
        const angle =
          ((i + ((layer + phases[group]) % 2) / 2) * Math.PI * 2) / count;
        positions.push({
          x: center.x + Math.cos(angle) * radiusX * width,
          y: center.y + Math.sin(angle) * radiusY * height,
        });
      }
    });
    return positions;
  });
}

// Explicit seeds restore exactly; retries request a different formation family.
/**
 * 根据关卡配置和种子生成完整棋盘。
 */
export function createLevel(
  levelIndex: number,
  layoutSeed?: number,
  previousLayoutSeed?: number,
  ingredientSeed?: number,
  ingredientPoolVersion: 1 | 2 = 2,
): LevelData {
  // 显式传入牌阵种子但未传食材种子时，保持旧版本的食材配比与顺序。
  if (ingredientSeed === undefined) {
    ingredientSeed =
      layoutSeed === undefined
        ? 1 + Math.floor(Math.random() * INGREDIENT_SEED_MAX)
        : 0;
  }
  if (layoutSeed === undefined) {
    layoutSeed =
      levelIndex === 2
        ? RANDOM_LAYOUT_SEED_START + Math.floor(Math.random() * 2147483646)
        : 0;
    if (levelIndex === 2
      && previousLayoutSeed !== undefined &&
      levelThreeLayoutFamily(layoutSeed) ===
        levelThreeLayoutFamily(previousLayoutSeed)) {
      layoutSeed =
        layoutSeed === RANDOM_LAYOUT_SEED_END
          ? RANDOM_LAYOUT_SEED_START
          : layoutSeed + 1;
    }
  }
  const config = LEVELS[levelIndex];
  const newLayout =
    config.level === 3 && layoutSeed >= RANDOM_LAYOUT_SEED_START;
  const random = seededRandom(
    newLayout ? layoutSeed : 20260923 + config.level * 97,
  );
  const pool = ingredientSeed
    ? selectLevelIngredients(levelIndex, ingredientSeed, ingredientPoolVersion)
    : undefined;
  const tiles: TileData[] = [];
  let id = 0;
  const stackedPositions =
    config.level === 3
      ? newLayout
        ? buildNewLevelThreePositions(config, layoutSeed)
        : buildLevelThreePositions(config, layoutSeed)
      : undefined;
  /**
   * 依次处理集合中的每个条目。
   */
  config.layers.forEach((count, layerIndex) => {
    const positions = stackedPositions
      ? stackedPositions[layerIndex]
      : buildPositions(count, layerIndex, config.rows);
    const types = buildTypes(
      count,
      config.typeCount,
      config.typeOffset || 0,
      layerIndex,
      random,
      pool,
    );
    for (let i = 0; i < count; i += 1) {
      tiles.push({
        id: `L${layerIndex}T${id++}`,
        type: types[i],
        x: positions[i].x,
        y: positions[i].y,
        layer: layerIndex,
        blockedBy: [],
        removed: false,
      });
    }
  });

  const tileScale = config.tileScale || (config.rows === 8 ? 0.82 : 1);
  const tileWidth = 92 * 1.2 * tileScale;
  const tileHeight = 76 * 1.2 * tileScale;
  // Only neighbouring spatial cells can overlap. Iterate uppers in original
  // order so saved layouts retain exactly the same ordered blocker lists.
  const cells = new Map<string, TileData[]>();
  for (const tile of tiles) {
    const key = `${Math.floor(tile.x / tileWidth)}:`
      + `${Math.floor(tile.y / tileHeight)}`;
    if (!cells.has(key)) {
      cells.set(key, []);
    }
    cells.get(key)!.push(tile);
  }
  for (const upper of tiles) {
    const cx = Math.floor(upper.x / tileWidth);
    const cy = Math.floor(upper.y / tileHeight);
    for (let dx = -1; dx <= 1; dx++) {
      for (let dy = -1; dy <= 1; dy++) {
        for (const lower of cells.get(`${cx + dx}:${cy + dy}`) || []) {
          if (upper.layer > lower.layer
            && Math.abs(lower.x - upper.x) < tileWidth
            && Math.abs(lower.y - upper.y) < tileHeight) {
            lower.blockedBy.push(upper.id);
          }
        }
      }
    }
  }
  return {
    config,
    tiles,
    total: tiles.length,
    ...(ingredientSeed ? {ingredientSeed} : {}),
    ...(ingredientSeed && ingredientPoolVersion === 2
      ? {ingredientPoolVersion} : {}),
    ...(config.level === 3 && layoutSeed !== 0 ? {layoutSeed} : {}),
  };
}

/**
 * 按食材编号查找显示配置。
 */
export function getIngredient(type: IngredientType): IngredientConfig {
  /**
   * 按条件筛选或判断条目：item.id === type。
   */
  return INGREDIENTS.find((item) => item.id === type) || INGREDIENTS[0];
}

/**
 * 只选择棋盘上未拿取的牌（含被遮挡的牌），不修改棋盘或托盘。
 */
export function selectBombTargets(
  tiles: TileData[],
  random: () => number = Math.random,
): TileData[] {
  const groups = new Map<IngredientType, TileData[]>();
  /**
   * 按条件筛选或判断条目：!tile.removed。
   */
  /**
   * 依次处理集合中的每个条目。
   */
  tiles
    .filter((tile) => !tile.removed)
    .forEach((tile) => {
      const group = groups.get(tile.type) || [];
      group.push(tile);
      groups.set(tile.type, group);
    });
  /**
   * 按条件筛选或判断条目：group.length >= 3。
   */
  const eligible = Array.from(groups.values()).filter(
    (group) => group.length >= 3,
  );
  if (eligible.length === 0) {
    return [];
  }
  const candidates = eligible[Math.floor(random() * eligible.length)].slice();
  const count = candidates.length >= 6 ? 6 : 3;
  return shuffle(candidates, random).slice(0, count);
}

  getItem(key: string): string | null;
}): PageStyle {

    const old = Number(storage.getItem('hotpot.page-style.v1'));
    const migrated: Record<number, PageStyle> = {1: 8, 2: 1, 7: 8, 9: 9, 10: 4};

  }
}

function recordProgress(
  progress: CollectionProgress,
  save: GameSave | null,
): CollectionProgress {
  const next = clone(progress);
  if (!save) {
    return next;
  }
  const removed = new Set(save.removedIds);
  const seen = new Set<IngredientType>([
    ...next.discovered,
    ...save.tray,
    ...save.reserve,
  ]);
  /**
   * 依次处理集合中的每个条目。
   */
  template(save.levelIndex).tiles.forEach((tile, index) => {
    if (removed.has(tile.id)) {
      seen.add(save.types[index]);
    }
  });
  /**
   * 将当前条目转换为后续处理所需的值。
   */
  /**
   * 按条件筛选或判断条目：seen.has(id)。
   */
  next.discovered = INGREDIENTS.map((item) => item.id).filter((id) =>
    seen.has(id),
  );
  const cleared =
    save.removedIds.length - save.tray.length - save.reserve.length;
  next.bestCleared[save.levelIndex] = Math.max(
    next.bestCleared[save.levelIndex],
    cleared,
  );
  if (save.status === 'won') {
    if (next.completedLevels.indexOf(save.levelIndex) < 0) {
      next.completedLevels.push(save.levelIndex);
    }
    if (!save.reviveUsed && next.cleanWins.indexOf(save.levelIndex) < 0) {
      next.cleanWins.push(save.levelIndex);
    }
  }
  return validateProgress(next);
}

/**
 * 使用关卡种子创建可验证的棋盘模板。
 */
function template(
  index: number,
  layoutSeed = 0,
  ingredientSeed = 0,
  ingredientPoolVersion: 1 | 2 = 1,
): LevelData {
  if (layoutSeed === 0 && ingredientSeed === 0) {
    if (!templates.has(index)) {
      templates.set(index, createLevel(index, 0, undefined, 0));
    }
    return templates.get(index)!;
  }
  // Inventory validation and geometry restoration use different seeds. They
  // must not evict each other and regenerate a full board on every restore.
  const key = `${index}:${layoutSeed}:${ingredientSeed}:${ingredientPoolVersion}`;
  if (!randomTemplates.has(key)) {
    randomTemplates.set(key,
      createLevel(index, layoutSeed, undefined, ingredientSeed,
        ingredientPoolVersion));
    if (randomTemplates.size > 4) {
      randomTemplates.delete(randomTemplates.keys().next().value);
    }
  }
  return randomTemplates.get(key)!;
}
/**
 * 检查稳定结算后的存档；不接受飞牌中或三消未结算的半成品。
 */
export function validateGameSave(value: any): GameSave {
  check(
    value &&
      value.version === SAVE_VERSION &&
      value.boardVersion === BOARD_VERSION,
    '存档版本不兼容',
  );
  check(integer(value.levelIndex, 0, LEVELS.length - 1), '关卡编号错误');
  check(
    value.runId === undefined || (typeof value.runId === 'string'
      && /^[a-z0-9-]{8,80}$/.test(value.runId)),
    '牌局标识错误',
  );
  check(
    value.layoutSeed === undefined ||
      (value.levelIndex === 2 &&
        integer(value.layoutSeed, 1, RANDOM_LAYOUT_SEED_END)),
    '牌阵种子错误',
  );
  check(
    value.ingredientSeed === undefined ||
      integer(value.ingredientSeed, 1, INGREDIENT_SEED_MAX),
    '食材种子错误',
  );
  check(value.ingredientPoolVersion === undefined
    || value.ingredientPoolVersion === 1 || value.ingredientPoolVersion === 2,
  '食材池版本错误');
  const level = template(value.levelIndex, 0, value.ingredientSeed,
    value.ingredientPoolVersion);
  check(
    Array.isArray(value.types) && value.types.length === level.total,
    '食材牌数错误',
  );
  const expected = new Map<string, number>();
  /**
   * 依次处理集合中的每个条目。
   */
  level.tiles.forEach((tile) =>
    expected.set(tile.type, (expected.get(tile.type) || 0) + 1),
  );
  /**
   * 依次处理集合中的每个条目。
   */
  value.types.forEach((type: string) => {
    check(expected.has(type), '食材种类错误');
    expected.set(type, expected.get(type)! - 1);
  });
  /**
   * 按条件筛选或判断条目：count === 0。
   */
  check(
    Array.from(expected.values()).every((count) => count === 0),
    '食材总数不守恒',
  );
  /**
   * 将当前条目转换为后续处理所需的值。
   */
  const ids = new Set(level.tiles.map((tile) => tile.id));
  const allowed = new Set(Array.from(expected.keys()) as IngredientType[]);
  /**
   * 检查 validateMove 对应的条件并报告不符合项。
   */
  const validateMove = (move: any, full: boolean): SavedMove => {
    check(
      move &&
        Array.isArray(move.removedIds) &&
        move.removedIds.length <= level.total &&
        /**
         * 按条件筛选或判断条目：ids.has(id)。
         */
        new Set(move.removedIds).size === move.removedIds.length &&
        move.removedIds.every((id: string) => ids.has(id)),
      '移除牌记录错误',
    );
    check(
      Array.isArray(move.tray) &&
        move.tray.length <= 7 &&
        Array.isArray(move.reserve) &&
        move.reserve.length <= 3 &&
        /**
         * 按条件筛选或判断条目：allowed.has(type)。
         */
        [...move.tray, ...move.reserve].every((type) => allowed.has(type)),
      '托盘或寄存区错误',
    );
    const counts = new Map<IngredientType, number>();
    const removed = new Set(move.removedIds);
    /**
     * 依次处理集合中的每个条目。
     */
    level.tiles.forEach((tile, i) => {
      if (!removed.has(tile.id)) {
        counts.set(value.types[i], (counts.get(value.types[i]) || 0) + 1);
      }
    });
    /**
     * 依次处理集合中的每个条目。
     */
    [...move.tray, ...move.reserve].forEach((type) =>
      counts.set(type, (counts.get(type) || 0) + 1),
    );
    /**
     * 按条件筛选或判断条目：count % 3 === 0。
     */
    check(
      Array.from(counts.values()).every((count) => count % 3 === 0),
      '剩余食材不守恒',
    );
    /**
     * 按条件筛选或判断条目：item === type。
     */
    for (const type of allowed)
      check(
        move.tray.filter((item: string) => item === type).length < 3,
        '托盘包含未结算三消',
      );
    if (!full) {
      check(move.tray.length < 7, '撤回快照托盘已满');
    }
    return {
      tray: move.tray.slice(),
      reserve: move.reserve.slice(),
      removedIds: move.removedIds.slice(),
    };
  };
  const move = validateMove(value, true);
  check(
    typeof value.reviveUsed === 'boolean' &&
      (value.reviveUsed || move.reserve.length === 0),
    '复活状态错误',
  );
  const maxRewards = value.levelIndex === 2 ? 8 : 0;
  check(integer(value.rewardedMilestones, 0, maxRewards), '阶段奖励错误');
  const cleared =
    move.removedIds.length - move.tray.length - move.reserve.length;
  check(
    cleared >= 0 &&
      cleared % 3 === 0 &&
      value.rewardedMilestones * 100 <= cleared + 3,
    '消除记录错误',
  );
  const toolUses = value.toolUses;
  /**
   * 按条件筛选或判断条目：integer(toolUses[type], 0, value.levelIndex === 0 ? 0 : 9)。
   */
  check(
    toolUses &&
      ['undo', 'bomb', 'shuffle'].every((type) =>
        integer(toolUses[type], 0, value.levelIndex === 0 ? 0 : 9),
      ),
    '道具次数错误',
  );
  check(
    toolUses.undo + toolUses.bomb + toolUses.shuffle <=
      (value.levelIndex === 0 ? 0 : 3) + value.rewardedMilestones,
    '道具次数超过已获奖励',
  );
  const empty = cleared === level.total;
  check(['playing', 'lost', 'won'].indexOf(value.status) >= 0, '对局状态错误');
  check(
    value.status === 'won'
      ? empty
      : value.status === 'lost'
        ? move.tray.length === 7
        : !empty && move.tray.length < 7,
    '对局状态与棋盘不一致',
  );
  let lastMove: SavedMove | null = null;
  if (value.lastMove !== null) {
    lastMove = validateMove(value.lastMove, false);
    const removed = new Set(move.removedIds);
    /**
     * 按条件筛选或判断条目：removed.has(id)。
     */
    check(
      lastMove.removedIds.every((id) => removed.has(id)),
      '撤回记录包含未来拿牌',
    );
    check(value.reviveUsed || lastMove.reserve.length === 0, '撤回寄存区错误');
  }
  return {
    ...move,
    ...(value.runId === undefined ? {} : {runId: value.runId}),
    version: 1,
    boardVersion: BOARD_VERSION,
    levelIndex: value.levelIndex,
    types: value.types.slice(),
    ...(value.layoutSeed === undefined ? {} : {layoutSeed: value.layoutSeed}),
    ...(value.ingredientSeed === undefined
      ? {}
      : {ingredientSeed: value.ingredientSeed}),
    ...(value.ingredientPoolVersion === undefined
      ? {} : {ingredientPoolVersion: value.ingredientPoolVersion}),
    reviveUsed: value.reviveUsed,
    rewardedMilestones: value.rewardedMilestones,
    toolUses: {
      undo: toolUses.undo,
      bomb: toolUses.bomb,
      shuffle: toolUses.shuffle,
    },
    lastMove,
    status: value.status,
  };
}

export function restoreLevel(save: GameSave): LevelData {

  const level = clone(
    template(valid.levelIndex, valid.layoutSeed, valid.ingredientSeed,
      valid.ingredientPoolVersion),

  /**
   * 依次处理集合中的每个条目。

    tile.removed = removed.has(tile.id);

    progress.regionChosen = true;

    /**
     * 将当前条目转换为后续处理所需的值。
     */
    profile: {regionCodes: path.map((item) => item.code)},
    save,
    progress: recordProgress(progress, save),
    playDate: validatePlayDate(value.playDate),
  };
}

import {RegionNode, check} from './PlayerTypes';

/**
 * 验证地区树的编码、名称和层级。
 */
export function validateRegions(value: unknown): RegionNode[] {
  const seen = new Set<string>();
  /**
   * 处理 walk 对应的游戏逻辑。
   */
  const walk = (items: any, depth: number): RegionNode[] => {
    check(
      Array.isArray(items) &&
        items.length > 0 &&
        items.length <= 200 &&
        depth <= 3,
      '地区数据格式错误',
    );
    /**
     * 将当前条目转换为后续处理所需的值。
     */
    return items.map((item) => {
      check(
        item &&
          typeof item.code === 'string' &&
          /^[A-Z0-9_]{1,20}$/.test(item.code) &&
          typeof item.name === 'string' &&
          item.name.length > 0 &&
          item.name.length <= 30 &&
          !seen.has(item.code),
        '地区节点格式错误',
      );
      seen.add(item.code);
      const node: RegionNode = {code: item.code, name: item.name};
      if (item.children !== undefined) {
        node.children = walk(item.children, depth + 1);
      }
      return node;
    });
  };
  return walk(value, 1);
}
/**
 * 根据地区编码解析地区树中的路径。
 */
export function regionPath(
  regions: RegionNode[],
  codes: unknown,
  allowEmpty = true,
): RegionNode[] {
  check(
    Array.isArray(codes) &&
      codes.length <= 3 &&
      (allowEmpty || codes.length > 0),
    '请选择完整地区',
  );
  const result: RegionNode[] = [];
  let nodes = regions;
  /**
   * 依次处理集合中的每个条目。
   */
  codes.forEach((code) => {
    /**
     * 按条件筛选或判断条目：item.code === code。
     */
    const selected = nodes.find((item) => item.code === code);
    check(selected, '地区代码或上下级关系错误');
    result.push(selected);
    nodes = selected.children || [];
  });
  check(codes.length === 0 || nodes.length === 0, '请选择完整地区');
  return result;
}

/** Read-only operational settings delivered by the player cloud function. */
export interface RuntimeConfig {
  collectionHeadingTaps: number;
  collectionIngredientTaps: number;
}

export const DEFAULT_RUNTIME_CONFIG: RuntimeConfig = {
  collectionHeadingTaps: 6,
  collectionIngredientTaps: 3,
};

export function normalizeRuntimeConfig(value: any): RuntimeConfig {
  const count = (key: keyof RuntimeConfig): number =>
    Number.isSafeInteger(value?.[key]) && value[key] >= 1 && value[key] <= 99
      ? value[key] : DEFAULT_RUNTIME_CONFIG[key];
  return {
    collectionHeadingTaps: count('collectionHeadingTaps'),
    collectionIngredientTaps: count('collectionIngredientTaps'),
  };
}
